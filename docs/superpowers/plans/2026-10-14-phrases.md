> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 ⌘⇧V 面板里加一个独立的「常用语」页：在面板里新建、编辑、删除、排序常用语，并像历史条目一样插入到原应用，不影响复制历史。

**Spec:** `docs/superpowers/specs/2026-10-14-phrases-design.md`

## Global Constraints

- Go 1.27.1；不新增 Go 模块或 npm 依赖；不用 cgo。
- 界面文案是中文，必须和 spec 一字不差（每个任务里都直接给出）。
- 单条常用语最多 65536 字节（64 KB）；最多 500 条；列表摘要 = 连续空白压缩成一个空格后取前 140 个字符，超出时末尾加「…」。
- 存储：`phrases.json` 和 `history.json` 同目录；文件 0600，目录 0700；格式 `{"version":1,"phrases":[{"id":"…","text":"…"}]}`；`id` 是 16 位小写十六进制（8 字节随机数）；内容原样保存。
- 错误文案（一字不差）：`内容不能为空`、`内容编码无效`、`单条常用语最多 64 KB`、`常用语最多 500 条`、`这条常用语已删除`、`常用语保存失败：<原因>`、`常用语文件无法读取，已停止修改：<路径>。修好或移走这个文件后重启 Clipnest。`
- 不加全局快捷键，不改菜单栏菜单。面板每次打开都显示历史页。
- 历史的行为不变：`Choose`、剪贴板监听、200 条 / 32 MB 上限、暂停、清空。
- 风格跟周围代码一致：Go 注释用英文，少而精；`web/*.js` 沿用 `app.js` 的紧凑写法；提交信息用约定式前缀（`feat(phrases): …` 等）。
- 在独立 worktree 的 `feat/phrases` 分支上做，基于最新的 `main`（Task 1 Step 1）。

```go
func (h *History) Save() error {
	h.writeMu.Lock()
	defer h.writeMu.Unlock()
	h.mu.Lock()
	if h.saved == h.version {
		h.mu.Unlock()
		return nil
	}
	a := archive{1, h.paused, append([]Entry{}, h.entries...)}
	version := h.version
	h.mu.Unlock()
	if err := writePrivateJSON(h.path, a); err != nil {
		return err
	}
	h.mu.Lock()
	h.saved = version
	h.mu.Unlock()
	return nil
}

// writePrivateJSON replaces path with v encoded as JSON, readable only by the
// user. Writing a temp file and renaming it means a crash never leaves half a file.
func writePrivateJSON(path string, v any) error {
	if err := os.MkdirAll(filepath.Dir(path), 0700); err != nil {
		return err
	}
	f, err := os.CreateTemp(filepath.Dir(path), "."+filepath.Base(path)+"-*")
	if err != nil {
		return err
	}
	name := f.Name()
	defer os.Remove(name)
	if err = f.Chmod(0600); err == nil {
		err = json.NewEncoder(f).Encode(v)
	}
	if err == nil {
		err = f.Sync()
	}
	closeErr := f.Close()
	if err == nil {
		err = closeErr
	}
	if err != nil {
		return err
	}
	return os.Rename(name, path)
}
```

新建 `phrases.go`：

```go
package main

import (
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"os"
	"slices"
	"strings"
	"sync"
	"unicode/utf8"
)

const maxPhrases = 500
const maxPhraseText = 64 << 10

// Phrase is text kept for reuse. IDs survive edits and the order is the
// user's own, so a phrase's number only changes when the user moves it.
type Phrase struct {
	ID      string `json:"id"`
	Text    string `json:"text,omitempty"`
	Excerpt string `json:"excerpt,omitempty"`
}

func (p Phrase) summary() Phrase {
	return Phrase{ID: p.ID, Excerpt: shorten(strings.Join(strings.Fields(p.Text), " "), 140)}
}

// Phrases makes a change visible only after it is on disk. A file that cannot
// be read turns the store read-only so user-written text is never replaced.
type Phrases struct {
	mu     sync.Mutex
	path   string
	items  []Phrase
	broken error
}
type phraseArchive struct {
	Version int      `json:"version"`
	Phrases []Phrase `json:"phrases"`
}

var errPhraseGone = errors.New("这条常用语已删除")

func checkPhraseText(text string) error {
	if strings.TrimSpace(text) == "" {
		return errors.New("内容不能为空")
	}
	if !utf8.ValidString(text) {
		return errors.New("内容编码无效")
	}
	if len(text) > maxPhraseText {
		return errors.New("单条常用语最多 64 KB")
	}
	return nil
}

func openPhrases(path string) *Phrases {
	p := &Phrases{path: path, items: []Phrase{}}
	items, err := readPhrases(path)
	if err != nil {
		p.broken = fmt.Errorf("常用语文件无法读取，已停止修改：%s。修好或移走这个文件后重启 Clipnest。", path)
		return p
	}
	p.items = items
	return p
}
func readPhrases(path string) ([]Phrase, error) {
	f, err := os.Open(path)
	if errors.Is(err, os.ErrNotExist) {
		return []Phrase{}, nil
	}
	if err != nil {
		return nil, err
	}
	defer f.Close()
	var a phraseArchive
	if err := json.NewDecoder(io.LimitReader(f, 256<<20)).Decode(&a); err != nil {
		return nil, err
	}
	if a.Version != 1 || len(a.Phrases) > maxPhrases {
		return nil, errors.New("unsupported phrases file")
	}
	items := []Phrase{}
	seen := map[string]bool{}
	for _, x := range a.Phrases {
		if x.ID == "" || seen[x.ID] {
			return nil, errors.New("missing or duplicate phrase id")
		}
		if err := checkPhraseText(x.Text); err != nil {
			return nil, err
		}
		seen[x.ID] = true
		items = append(items, Phrase{ID: x.ID, Text: x.Text})
	}
	return items, nil
}
func (p *Phrases) index(id string) int {
	return slices.IndexFunc(p.items, func(x Phrase) bool { return x.ID == id })
}

// List returns summaries in the user's order, filtered case-insensitively.
func (p *Phrases) List(q string) ([]Phrase, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	if p.broken != nil {
		return nil, p.broken
	}
	q = strings.ToLower(strings.TrimSpace(q))
	out := []Phrase{}
	for _, x := range p.items {
		if q == "" || strings.Contains(strings.ToLower(x.Text), q) {
			out = append(out, x.summary())
		}
	}
	return out, nil
}
func (p *Phrases) Get(id string) (Phrase, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	if p.broken != nil {
		return Phrase{}, p.broken
	}
	i := p.index(id)
	if i < 0 {
		return Phrase{}, errPhraseGone
	}
	return p.items[i], nil
}

// Save appends a new phrase when id is empty and otherwise replaces the text
// of an existing one in place.
func (p *Phrases) Save(id, text string) (Phrase, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	if p.broken != nil {
		return Phrase{}, p.broken
	}
	if err := checkPhraseText(text); err != nil {
		return Phrase{}, err
	}
	items := slices.Clone(p.items)
	saved := Phrase{ID: id, Text: text}
	if id == "" {
		if len(items) >= maxPhrases {
			return Phrase{}, errors.New("常用语最多 500 条")
		}
		saved.ID = p.newID()
		items = append(items, saved)
	} else if i := p.index(id); i >= 0 {
		items[i] = saved
	} else {
		return Phrase{}, errPhraseGone
	}
	if err := p.commit(items); err != nil {
		return Phrase{}, err
	}
	return saved, nil
}
func (p *Phrases) Delete(id string) error {
	p.mu.Lock()
	defer p.mu.Unlock()
	if p.broken != nil {
		return p.broken
	}
	i := p.index(id)
	if i < 0 {
		return errPhraseGone
	}
	return p.commit(slices.Delete(slices.Clone(p.items), i, i+1))
}

// Move shifts a phrase one place up (-1) or down (1). Moving past either end
// changes nothing and leaves the file alone.
func (p *Phrases) Move(id string, delta int) error {
	p.mu.Lock()
	defer p.mu.Unlock()
	if p.broken != nil {
		return p.broken
	}
	if delta != -1 && delta != 1 {
		return errors.New("无效的移动")
	}
	i := p.index(id)
	if i < 0 {
		return errPhraseGone
	}
	j := i + delta
	if j < 0 || j >= len(p.items) {
		return nil
	}
	items := slices.Clone(p.items)
```

```js
import { numberIndex } from './selection.js';

// Maps a keydown to the panel action it triggers, or null. app.js only performs
// the action, so the rules for both pages stay in one tested place.
export function keyAction(event, ctx) {
  if (event.isComposing || event.keyCode === 229) return null;
  const key = event.key, lower = key.toLowerCase(), command = event.metaKey || event.ctrlKey, phrases = ctx.view === 'phrases';
  if (lower === 'v' && command && event.shiftKey && ctx.demo) return { type: 'reopen' };
  if (ctx.dialogOpen) return null;
  if (key === 'Escape') return { type: ctx.hasQuery ? 'clear-search' : 'dismiss' };
  if (command && lower === 'k') return { type: 'search' };
  if (ctx.busy) return null;
  if (command && !event.altKey && !event.shiftKey && phrases) {
    if (lower === 'n') return { type: 'new' };
    if (lower === 'e' && ctx.hasSelection) return { type: 'edit' };
  }
  if (event.altKey && !command && !event.shiftKey && (key === 'ArrowUp' || key === 'ArrowDown')) {
    return phrases && !ctx.hasQuery && !ctx.typing ? { type: 'reorder', delta: key === 'ArrowDown' ? 1 : -1 } : null;
  }
  if (command || event.altKey || event.shiftKey) return null;
  if (key === 'ArrowDown' || key === 'ArrowUp') return { type: 'select', delta: key === 'ArrowDown' ? 1 : -1 };
  if ((key === 'ArrowLeft' || key === 'ArrowRight') && !ctx.typing) return { type: 'view', view: key === 'ArrowLeft' ? 'history' : 'phrases' };
  if (key === 'Enter' && (ctx.typing || ctx.atList)) return { type: 'choose' };
  const index = ctx.typing || event.repeat ? -1 : numberIndex(key, ctx.count);
  return index >= 0 ? { type: 'number', index } : null;
}
```
