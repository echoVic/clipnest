# 钉住面板 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 面板可以钉住：失焦不收起，粘贴后不收起，切换应用时粘贴目标跟着变，新复制的内容实时出现。

**Architecture:** 给现有面板加 `Desktop.pinned`，只在主线程读写，`hidden()` 清掉它，所以所有收起路径都结束钉住。失焦、快捷键、目标跟踪的判断抽成纯函数测试。粘贴、打开链接、开始顺序粘贴在钉住时改走 `leavePanel`：交还键盘，不收起。新复制发 `clipnest:history-changed`，由前端在空闲时调 `Reload`；后端在后台换好新快照，并找出选中条目的新位置，一起返回。

**Tech Stack:** Go、MyGo 0.9.0（Window.Blur/IsFocused/ShowInactive、事件）、SQLite 只读事务、原生 JS、Node test runner。

**Spec:** `docs/superpowers/specs/2026-11-17-pin-panel-design.md`

## Global Constraints

- 直接在 `feat/pin-panel` 分支按顺序执行，不用 git worktree；每个任务做完就提交。
- 没钉住时面板的行为完全不变。
- 快捷键 ⌘P（Windows、Linux 为 Ctrl+P），动作名 `pin`。
- 按钮文案：未钉住「钉住面板（⌘P）」，钉住「取消钉住（⌘P）」；读屏文字去掉括号部分。
- 会话过期的报错沿用「面板已更新，请重新选择」。
- `Reload` 不广播；`historyView.Locate` 最多找 1000 条，每页 200 条，找不到返回 -1。
- 前端刷新的重试间隔 300 ms。
- 绝不把粘贴按键发给钉住的面板自己。
- 键名占位符（如 `ArrowDown`）照原样写，不改写成真实键名。

```go
package main

// hideOnBlur reports whether the panel hides when it loses the keyboard. A
// pinned panel stays, as does one busy pasting or guiding a permission.
func hideOnBlur(busy, permission, pinned bool) bool {
	return !busy && !permission && !pinned
}

type toggleAction int

const (
	toggleOpen toggleAction = iota
	toggleFocus
	toggleClose
)

// panelToggle decides what the summon shortcut and the tray button do. A
// pinned panel without the keyboard takes it back instead of hiding.
func panelToggle(visible, pinned, focused bool) toggleAction {
	switch {
	case !visible:
		return toggleOpen
	case pinned && !focused:
		return toggleFocus
	default:
		return toggleClose
	}
}

// trackedTarget is the app a pinned panel pastes into: the last app in front
// that is not Clipnest, which Target reports as PID 0.
func trackedTarget(current, front targetApp) targetApp {
	if front.PID == 0 {
		return current
	}
	return front
}
```

```go
// sequenceOwnWindow reports whether a press of the paste keys belongs to one

// takes without making Clipnest the front app, or Clipnest is in front. A
// pinned panel shows without the keyboard, so showing is not enough.
func sequenceOwnWindow(panelFocused bool, front targetApp) bool {
```

```go
// locateLimit bounds how far Locate reads; a variable so tests can lower it.
var locateLimit = 1000

// Locate returns where the entry id sits in the list for q and kind, or -1
// when it is not within the first locateLimit entries.
func (v *historyView) Locate(q, kind, id string) (int, error) {
	for offset := 0; offset < locateLimit; {
		page, err := v.Page(q, kind, offset, min(200, locateLimit-offset))
		if err != nil {
			return -1, err
		}
		for i, e := range page.Entries {
			if e.ID == id {
				return offset + i, nil
			}
		}
		offset += len(page.Entries)
		if len(page.Entries) == 0 || offset >= page.Total {
			break
		}
	}
	return -1, nil
}
```

```go
// historyChanged tells a showing panel, by session, that its snapshot is
// behind. The panel asks for Reload when nothing holds the old snapshot.
var historyChanged = mygo.NewEvent[uint64]("clipnest:history-changed")

func (d *Desktop) announceHistory() {
	if !d.quitting && d.win != nil && d.win.IsVisible() {
		_ = historyChanged.Broadcast(d.session)
	}
}

// PanelReload is a panel with a new snapshot and where the selected entry is
// in it, -1 when it is gone or none was given.
type PanelReload struct {
	State PanelState `json:"state"`
	Index int        `json:"index"`
}

// Reload swaps a showing panel's snapshot for a new one. The new snapshot is
// opened and read in the background while the old one keeps serving the
// panel; only the swap runs on main. It does not broadcast: the panel takes
// a broadcast of a new session as reopening and clears its search.
func (d *Desktop) Reload(session uint64, q, kind, id string) (r PanelReload, err error) {
	r.Index = -1
	ready := false
	mygo.RunOnMain(func() {
		ready = session == d.session && d.win != nil && d.win.IsVisible() && !d.busy && !d.quitting && d.view != nil
		if !ready {
			r.State = d.snapshot()
		}
	})
	if !ready {
		return r, nil
	}
	ctx, cancel := context.WithCancel(context.Background())
	v, loadErr := d.h.openView(ctx)
	var page HistoryPage
	if loadErr == nil {
		page, loadErr = v.Page("", "", 0, historyPageSize)
	}
	index := -1
	if loadErr == nil && id != "" {
		// A failed lookup only loses the selection; the panel falls back to
		// the old position.
		if found, err := v.Locate(q, kind, id); err == nil {
			index = found
		}
	}
	mygo.RunOnMain(func() {
		if loadErr != nil || session != d.session || !d.win.IsVisible() || d.busy || d.quitting {
			cancel()
			if v != nil {
				go v.Close()
			}
			if loadErr != nil {
				d.notice = "历史无法读取：" + loadErr.Error()
			}
			r.State = d.snapshot()
			return
		}
		d.closeView()
		d.session++
		d.view, d.cancelView = v, cancel
		d.entries, d.total, d.loading = page.Entries, page.Total, false
		if strings.HasPrefix(d.notice, "历史无法读取：") {
			d.notice = ""
		}
		r.State, r.Index = d.snapshot(), index
	})
	return r, nil
}
```

```go
func (d *Desktop) SetPinned(session uint64, pinned bool) (s PanelState, err error) {
	mygo.RunOnMain(func() {
		if session != d.session || d.win == nil || !d.win.IsVisible() || d.quitting {
			err = errors.New("面板已更新，请重新选择")
			return
		}
		d.pinned = pinned
		s = d.snapshot()
	})
	return
}

// trackTarget follows the app a pinned panel pastes into as the user
// switches apps, and shows its name in the panel when it changes.
func (d *Desktop) trackTarget() {
	if !d.pinned || d.win == nil || !d.win.IsVisible() {
		return
	}
	next := trackedTarget(d.target, d.n.Target())
	changed := next.Name != d.target.Name
	d.target = next
	if changed {
		_ = panelChanged.Broadcast(d.snapshot())
	}
}

// leavePanel gives the keyboard back to target: a pinned panel stays up,
// any other one hides.
func (d *Desktop) leavePanel(target targetApp) {
	if d.pinned {
		d.handOff()
	} else {
		d.hide()
	}
	d.n.Activate(target)
}

// handOff takes the keyboard from a pinned panel without hiding it. On macOS
// the panel takes the keys without activating Clipnest, so activating the
// app in front does not take them back; ordering the panel out does, and
// showing it again without focus keeps it on screen. handingOff keeps that
// brief hide from ending the session. On Windows, Activate hands over the
// foreground and the panel stays above it.
func (d *Desktop) handOff() {
	if runtime.GOOS != "darwin" {
		return
	}
	d.handingOff = true
	d.win.Hide()
	d.win.ShowInactive()
	d.handingOff = false
}
```

- 新增：

```js
// A copy while the panel shows. Swap in the new snapshot only while nothing
// holds the old one: on the history page, no dialog or menu, no paste or load.
let reloading = false, reloadTimer = 0, heldPanel = null;
const reloadable = () => view === 'history' && !busy && !choosing && !migrating && !state.loading && !queryPending && !$('dialog[open]');
async function reloadHistory() {
  clearTimeout(reloadTimer); reloadTimer = 0;
  if (reloading) return;
  if (!reloadable()) { reloadTimer = setTimeout(reloadHistory, 300); return; }
  reloading = true;
  const session = state.session, list = $('#list'), index = pages.selected, q = $('#search').value;
  // At the top, follow the newest copy; anywhere else, keep the entry.
  const follow = index <= 0 && list.scrollTop === 0, offset = index * ROW_HEIGHT - list.scrollTop;
  try {
    const { state: next, index: found } = await call('Reload', session, q, kind, follow ? '' : currentEntry()?.id ?? '');
    if (next.session !== session && session === state.session) {
      restore = follow ? { index: 0, q, kind } : found >= 0 ? { index: found, q, kind, top: Math.max(0, found * ROW_HEIGHT - offset) } : { index, q, kind };
    }
    applyState(next);
  } catch (err) { toast(message(err)); }
  finally {
    reloading = false;
    const held = heldPanel; heldPanel = null;
    if (held) applyState(held);
  }
}
api.on('clipnest:history-changed', session => { if (session === state.session) reloadHistory(); });
```
